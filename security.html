Blacklisted IPs table create table if not exists blocked_ips ( ip_address text primary key, reason text, blocked_at timestamptz default now() ); revoke all on blocked_ips from anon, authenticated; -- 2. 1-Click Ban RPC create or replace function ban_attacker_ip_secure( p_token text, p_ip text, p_reason text ) returns json language plpgsql security definer set search_path = public, extensions as $$ declare v_user erp_users%rowtype; begin v_user := internal_verify_session(p_token); if v_user.role <> 'admin' then raise exception 'Access Denied: Only Administrators can blacklist IP addresses.'; end if; insert into blocked_ips (ip_address, reason) values (p_ip, p_reason) on conflict (ip_address) do update set reason = excluded.reason; insert into erp_audit_logs (module, action, record_identifier, details, performed_by) values ( 'Security Shield', 'IP_BANNED', p_ip, 'IP Blacklisted via 1-Click Sentinel Action: ' || p_reason, v_user.username ); return json_build_object('success', true, 'ip', p_ip); end; $$; -- 3. 1-Click Unban RPC create or replace function unban_ip_secure( p_token text, p_ip text ) returns json language plpgsql security definer set search_path = public, extensions as $$ declare v_user erp_users%rowtype; begin v_user := internal_verify_session(p_token); if v_user.role <> 'admin' then raise exception 'Access Denied: Only Administrators can unblock IP addresses.'; end if; delete from blocked_ips where ip_address = p_ip; insert into erp_audit_logs (module, action, record_identifier, details, performed_by) values ( 'Security Shield', 'IP_UNBANNED', p_ip, 'IP Removed from Blacklist via Sentinel Dashboard', v_user.username ); return json_build_object('success', true, 'ip', p_ip); end; $$;
